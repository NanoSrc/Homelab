   no shutdown

! VLAN 30 - CLIENTS
! Ethernet29-33

   switchport access vlan 30
   no shutdown

! ============================================================

!

interface Ethernet34
   description UNIFI-PRO-MAX-UPLINK
   switchport mode trunk
   switchport trunk allowed vlan 10,20,30,40
   no shutdown

! VLAN 40 - IOT
